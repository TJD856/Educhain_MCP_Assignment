# 2. Install required dependencies

# 3. (Optional) If you need to upgrade a package (e.g., openai)
pip install --upgrade openai

# 4. Run the FastAPI MCP server (default port 5000)
uvicorn server:app --reload --port 5000

# 5. Test the /mcq endpoint using PowerShell (Invoke-RestMethod)
$body = @{ "topic" = "Photosynthesis"; "num" = 3 }
Invoke-RestMethod -Uri "http://127.0.0.1:5000/mcq" -Method Post -Body ($body | ConvertTo-Json) -ContentType "application/json"

# 6. Test the /lesson endpoint using PowerShell (Invoke-RestMethod)
$body = @{ "topic" = "Photosynthesis" }
Invoke-RestMethod -Uri "http://127.0.0.1:5000/lesson" -Method Post -Body ($body | ConvertTo-Json) -ContentType "application/json"

# 7. (Alternative) Test endpoints using curl (in Command Prompt or Git Bash)
# For /mcq endpoint:
curl -X POST "http://127.0.0.1:5000/mcq" -H "Content-Type: application/json" -d "{\"topic\": \"Photosynthesis\", \"num\": 3}"
# For /lesson endpoint:
curl -X POST "http://127.0.0.1:5000/lesson" -H "Content-Type: application/json" -d "{\"topic\": \"Photosynthesis\"}"
